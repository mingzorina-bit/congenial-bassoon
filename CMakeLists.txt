cmake_minimum_required(VERSION 3.24)
project(BilingualInput VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8 /W4)
endif()
add_library(BilingualCore STATIC src/BilingualCore/InputSession.cpp)
target_include_directories(BilingualCore PUBLIC src/BilingualCore)
add_executable(CoreTests tests/CoreTests/CoreTests.cpp)
target_link_libraries(CoreTests PRIVATE BilingualCore)
enable_testing()
add_test(NAME CoreTests COMMAND CoreTests)
option(BI_WITH_RIME "Build real Rime adapter and native WinUI bridge" OFF)
if(BI_WITH_RIME)
  if(NOT EXISTS "${RIME_ROOT}/include/rime_api.h")
    message(FATAL_ERROR "Run scripts/Prepare-Rime.ps1 first")
  endif()
  add_library(RimeAdapter STATIC src/BilingualCore/RimeAdapter.cpp)
  target_include_directories(RimeAdapter PUBLIC "${RIME_ROOT}/include")
  target_compile_definitions(RimeAdapter PUBLIC RIME_IMPORTS)
  target_link_libraries(RimeAdapter PUBLIC BilingualCore "${RIME_ROOT}/lib/rime.lib")
  add_library(BilingualNative SHARED src/BilingualCore/NativeApi.cpp)
  target_link_libraries(BilingualNative PRIVATE RimeAdapter)
  add_executable(RimeTests tests/CoreTests/RimeTests.cpp)
  target_link_libraries(RimeTests PRIVATE RimeAdapter)
  foreach(target BilingualNative RimeTests)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${RIME_ROOT}/lib/rime.dll" "$<TARGET_FILE_DIR:${target}>/rime.dll")
  endforeach()
  add_test(NAME RimeTests COMMAND RimeTests "${CMAKE_SOURCE_DIR}/data/rime" "${CMAKE_BINARY_DIR}/rime-test-user")
endif()
